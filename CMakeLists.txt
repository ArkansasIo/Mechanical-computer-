cmake_minimum_required(VERSION 3.20)
project(GearLogic1024 VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(gearlogic src/GearLogic.cpp)
target_include_directories(gearlogic PUBLIC include)
if(MSVC)
 target_compile_options(gearlogic PRIVATE /W4 /permissive-)
else()
 target_compile_options(gearlogic PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(gearlogic_demo src/main.cpp)
target_link_libraries(gearlogic_demo PRIVATE gearlogic)
enable_testing()
add_executable(gearlogic_tests tests/test.cpp)
target_link_libraries(gearlogic_tests PRIVATE gearlogic)
add_test(NAME gearlogic_tests COMMAND gearlogic_tests)
